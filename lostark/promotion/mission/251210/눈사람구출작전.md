``` javascript
(function () {

    if (window.__snowmanAutoV6) return;
    window.__snowmanAutoV6 = true;

    const DOORS_PER_FLOOR = 4;

    function sleep(ms) {
        return new Promise(r => setTimeout(r, ms));
    }

    function detectTotalFloors() {
        const stage = document.querySelector(".stage");
        if (!stage) return 1;

        return Array.from(stage.children)
            .filter(el =>
                el.className &&
                el.className.startsWith("floor") &&
                !isNaN(Number(el.className.slice(5)))
            )
            .length || 1;
    }

    let TOTAL_FLOORS = detectTotalFloors();
    let FLOORS = Array.from({ length: TOTAL_FLOORS }, (_, i) => i + 1);

    const snowCorrect = window.__snowCorrect = {};
    const snowWrong = window.__snowWrong = {};

    function initMemory() {
        FLOORS.forEach(f => {
            snowCorrect[f] = null;
            snowWrong[f] = new Set();
        });
    }

    let stopFlag = false;
    let autoRunning = false;
    let lastRemainCount = null;

    initMemory();

    function resetMemory() {
        TOTAL_FLOORS = detectTotalFloors();
        FLOORS = Array.from({ length: TOTAL_FLOORS }, (_, i) => i + 1);
        initMemory();
        lastRemainCount = null;
    }

    function floorFromIndex(idx) {
        return Math.floor(idx / DOORS_PER_FLOOR) + 1;
    }

    function floorRange(floor) {
        const start = (floor - 1) * DOORS_PER_FLOOR;
        return { start, end: start + DOORS_PER_FLOOR - 1 };
    }

    function detectFloorByCharacter() {
        for (let f of FLOORS) {
            const el = document.querySelector(".floor" + f);
            if (el && el.querySelector(".obj_character")) return f;
        }
        return null;
    }

    function detectFloorByActiveDoor() {
        for (let f of FLOORS) {
            const el = document.querySelector(".floor" + f);
            if (!el) continue;

            const active = Array.from(el.querySelectorAll("button.door"))
                .filter(btn => !btn.disabled);
            if (active.length > 0) return f;
        }
        return null;
    }

    function detectFloor() {
        return (
            detectFloorByCharacter() ||
            detectFloorByActiveDoor() ||
            FLOORS[0]
        );
    }

    async function waitFloorReady(floor, timeout = 3000) {
        const start = Date.now();

        while (Date.now() - start < timeout) {
            const charFloor = detectFloorByCharacter();
            const el = document.querySelector(".floor" + floor);
            const active = el
                ? Array.from(el.querySelectorAll("button.door")).filter(b => !b.disabled)
                : [];

            if (charFloor === floor && active.length > 0) return true;
            await sleep(80);
        }
        return false;
    }

    function clickDoor(idx) {
        const btn = document.querySelector('button.door[data-idx="' + idx + '"]');
        if (btn && !btn.disabled) {
            btn.click();
            return true;
        }
        return false;
    }

    function scanPassedDoors() {
        FLOORS.forEach(f => {
            const el = document.querySelector(".floor" + f);
            if (!el) return;

            const passed = el.querySelector("button.door--passed");
            if (passed) {
                const idx = Number(passed.dataset.idx);
                if (!isNaN(idx)) snowCorrect[f] = idx;
            }
        });
    }

    function applyPositionIndexes(str) {
        if (!str) return;
        str.split(",").map(Number).forEach((idx, i) => {
            const f = i + 1;
            if (snowCorrect[f] == null) snowCorrect[f] = idx;
        });
    }

    function applyHintIndexes(str) {
        if (!str) return;
        str.split(",").map(Number).forEach((idx, i) => {
            const f = i + 1;
            if (snowCorrect[f] == null) snowCorrect[f] = idx;
        });
    }

    function processResponse(clickedIdx, json) {
        if (typeof clickedIdx !== "number") return;

        const floor = floorFromIndex(clickedIdx);
        lastRemainCount = json.remainCount;

        if (json.positionIndexes) applyPositionIndexes(json.positionIndexes);
        if (json.hintIndexes) applyHintIndexes(json.hintIndexes);

        if (json.isCorrect) {
            snowCorrect[floor] = clickedIdx;
        } else {
            snowWrong[floor].add(clickedIdx);
        }

        if (json.isComplete) resetMemory();
    }

    (function hookXHR() {
        if (window.__snowmanXHRHooked) return;
        window.__snowmanXHRHooked = true;

        const OriginalXHR = window.XMLHttpRequest;

        function WrappedXHR() {
            const xhr = new OriginalXHR();
            let url = null;
            let body = null;

            const open = xhr.open;
            xhr.open = function (m, u) {
                url = u;
                return open.apply(xhr, arguments);
            };

            const send = xhr.send;
            xhr.send = function (data) {
                body = data;

                xhr.addEventListener("load", () => {
                    try {
                        if (!url || url.indexOf("SetDoor") === -1) return;

                        let idx = null;
                        if (typeof body === "string") {
                            const pos = body.indexOf("index=");
                            if (pos !== -1) {
                                idx = Number(body.slice(pos + 6));
                            }
                        }

                        processResponse(idx, JSON.parse(xhr.responseText));
                    } catch { }
                });

                return send.apply(xhr, arguments);
            };

            return xhr;
        }

        WrappedXHR.prototype = OriginalXHR.prototype;
        window.XMLHttpRequest = WrappedXHR;
    })();

    async function playOneStep() {
        if (stopFlag) return false;

        scanPassedDoors();

        const floor = detectFloor();
        await waitFloorReady(floor);

        const { start, end } = floorRange(floor);

        if (snowCorrect[floor] != null) {
            if (clickDoor(snowCorrect[floor])) return true;
            snowCorrect[floor] = null;
        }

        for (let i = start; i <= end; i++) {
            if (snowWrong[floor].has(i)) continue;
            if (clickDoor(i)) return true;
        }

        for (let i = start; i <= end; i++) {
            if (clickDoor(i)) return true;
        }

        return false;
    }

    async function autoLoop() {
        if (autoRunning) return;
        autoRunning = true;

        while (!stopFlag) {
            try { await playOneStep(); } catch { }
            await sleep(250);
        }

        autoRunning = false;
    }

    window.addEventListener("keydown", e => {
        if (e.key === "Escape") stopFlag = true;
    });

    autoLoop();

})();
```
